module uart_receiver (
    input        clk,
    input        rst_n,
    input        rx,
    input        rx_en,

    output reg [7:0] rx_data,
    output reg       rx_done
);

    //==========================================================
    // FSM STATES
    //==========================================================
    parameter IDLE_STATE  = 3'b000;
    parameter START_STATE = 3'b001;
    parameter DATA_STATE  = 3'b010;
    parameter STOP_STATE  = 3'b011;
    parameter DONE_STATE  = 3'b100;

    //==========================================================
    // INTERNAL REGISTERS
    //==========================================================
    reg [2:0] state;
    reg [3:0] sample;
    reg [2:0] index;
    reg [7:0] temp_register;

    //==========================================================
    // UART RECEIVER
    //==========================================================
    always @(posedge clk or negedge rst_n)
    begin
        if (!rst_n)
        begin
            state        <= IDLE_STATE;
            sample       <= 4'd0;
            index        <= 3'd0;
            temp_register <= 8'd0;
            rx_data      <= 8'd0;
            rx_done      <= 1'b0;
        end
        else
        begin
            // rx_done is normally low
            rx_done <= 1'b0;

            // Receiver operates only on rx_en tick
            if (rx_en)
            begin
                case (state)

                    //==================================================
                    // IDLE
                    //==================================================
                    IDLE_STATE:
                    begin
                        sample <= 4'd0;

                        // Detect possible START bit
                        if (rx == 1'b0)
                        begin
                            state  <= START_STATE;
                            sample <= 4'd0;
                        end
                    end

                    //==================================================
                    // START
                    // Verify middle of START bit
                    //==================================================
                    START_STATE:
                    begin
                        sample <= sample + 1'b1;

                        // 8th sample = middle of START bit
                        if (sample == 4'd7)
                        begin
                            if (rx == 1'b0)
                            begin
                                // Valid start bit
                                state         <= DATA_STATE;
                                sample        <= 4'd0;
                                index         <= 3'd0;
                                temp_register <= 8'd0;
                            end
                            else
                            begin
                                // False start
                                state  <= IDLE_STATE;
                                sample <= 4'd0;
                            end
                        end
                    end

                    //==================================================
                    // DATA
                    // Receive 8 bits LSB first
                    //==================================================
                    DATA_STATE:
                    begin
                        sample <= sample + 1'b1;

                        // Middle of each DATA bit
                        if (sample == 4'd15)
                        begin
                            temp_register[index] <= rx;
                            sample <= 4'd0;

                            if (index == 3'd7)
                            begin
                                // All 8 data bits received
                                state <= STOP_STATE;
                            end
                            else
                            begin
                                index <= index + 1'b1;
                            end
                        end
                    end

                    //==================================================
                    // STOP
                    //==================================================
                    STOP_STATE:
                    begin
                        sample <= sample + 1'b1;

                        if (sample == 4'd15)
                        begin
                            if (rx == 1'b1)
                            begin
                                // Valid stop bit
                                state <= DONE_STATE;
                            end
                            else
                            begin
                                // Invalid stop bit
                                state <= IDLE_STATE;
                            end

                            sample <= 4'd0;
                        end
                    end

                    //==================================================
                    // DONE
                    //==================================================
                    DONE_STATE:
                    begin
                        rx_data <= temp_register;
                        rx_done <= 1'b1;

                        state  <= IDLE_STATE;
                        sample <= 4'd0;
                        index  <= 3'd0;
                    end

                    //==================================================
                    // DEFAULT
                    //==================================================
                    default:
                    begin
                        state  <= IDLE_STATE;
                        sample <= 4'd0;
                        index  <= 3'd0;
                    end

                endcase
            end
        end
    end

endmodule
